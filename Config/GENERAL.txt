# ---- UPnP 依赖修复 ----
# 原始配置没有编译 iptables 相关内核模块，在线安装 luci-app-upnp 时会报：
#   “依赖的软件包 kmod-ip6tables / kmod-nft-compat / kmod-ipt-core 在所有仓库都未提供”
# 原因：默认防火墙是 nftables(fw4)，iptables 的内核模块没有被选中，
#       因此固件对应的软件源里根本没有这三个 kmod。
# 处理：把 iptables-nft / ip6tables-nft 及其内核依赖一起编译，
#       同时直接内置 UPnP（nftables 变体，运行时不依赖 iptables）。
CONFIG_PACKAGE_kmod-ipt-core=y
CONFIG_PACKAGE_kmod-ip6tables=y
CONFIG_PACKAGE_kmod-nft-compat=y
CONFIG_PACKAGE_iptables-nft=y
CONFIG_PACKAGE_ip6tables-nft=y
CONFIG_PACKAGE_miniupnpd-nftables=y
CONFIG_PACKAGE_luci-app-upnp=y
